set(BENCHMARK_NAME hapi_kernel_launch)
set(BENCHMARK_SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/hapi_kernel_launch.C)
set(BENCHMARK_INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/hapi_kernel_launch.ci)

find_program(CHARMC_EXECUTABLE NAMES charmc HINTS "${CHARM_ROOT}/bin" REQUIRED)

set(BACKEND_DEFINE "")
set(BACKEND_CHARM_FLAG "")
if(BENCHMARK_BACKEND STREQUAL "CUDA")
  set(BACKEND_DEFINE "-DBENCHMARK_USE_CUDA")
  set(BACKEND_CHARM_FLAG "-cuda")
elseif(BENCHMARK_BACKEND STREQUAL "HIP")
  set(BACKEND_DEFINE "-DBENCHMARK_USE_HIP")
  set(BACKEND_CHARM_FLAG "-hip")
endif()

set(KOKKOS_INCLUDE_DIR "${KOKKOS_ROOT}/include")
set(KOKKOS_LIBRARY_DIR "${KOKKOS_ROOT}/lib")

if(NOT EXISTS "${KOKKOS_INCLUDE_DIR}")
  message(FATAL_ERROR "Kokkos include directory not found: ${KOKKOS_INCLUDE_DIR}")
endif()

if(NOT EXISTS "${KOKKOS_LIBRARY_DIR}")
  message(FATAL_ERROR "Kokkos library directory not found: ${KOKKOS_LIBRARY_DIR}")
endif()

add_custom_command(
  OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/hapi_kernel_launch.decl.h ${CMAKE_CURRENT_BINARY_DIR}/hapi_kernel_launch.def.h
  COMMAND ${CHARMC_EXECUTABLE} ${BENCHMARK_INTERFACE}
  DEPENDS ${BENCHMARK_INTERFACE}
  WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
  VERBATIM
)

add_custom_command(
  OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/${BENCHMARK_NAME}
  COMMAND ${CHARMC_EXECUTABLE}
          ${BACKEND_CHARM_FLAG}
          -language charm++
          ${BACKEND_DEFINE}
          -I${KOKKOS_INCLUDE_DIR}
          -L${KOKKOS_LIBRARY_DIR}
          -lkokkoscore
          -I${CMAKE_CURRENT_BINARY_DIR}
          -o ${CMAKE_CURRENT_BINARY_DIR}/${BENCHMARK_NAME}
          ${BENCHMARK_SOURCE}
  DEPENDS
          ${BENCHMARK_SOURCE}
          ${CMAKE_CURRENT_BINARY_DIR}/hapi_kernel_launch.decl.h
          ${CMAKE_CURRENT_BINARY_DIR}/hapi_kernel_launch.def.h
  WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
  VERBATIM
)

add_custom_target(${BENCHMARK_NAME} ALL DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/${BENCHMARK_NAME})
